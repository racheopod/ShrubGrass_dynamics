//Create buffer function for springs, wells, and waterbodies
var buffer2km = function(thisFeature){
  return thisFeature.buffer(2000);
};

//Create buffer function for perennial streams + canals (1 km):
var buffer1km = function(thisFeature){
  return thisFeature.buffer(1000);
};

//Create buffer function for roads, rails, and intermittent streams
var buffer50m = function(thisFeature){
  return thisFeature.buffer(50);
};

//Buffer features:
var buffsprings = springs.map(buffer2km);
var buffwells = wells.map(buffer2km);
var buffwaterbodies = waterbodies.map(buffer2km);

var buffperstreams = perstreams.map(buffer1km);
var buffcanals = canals.map(buffer1km);

var buffintstreams1 = intstreams1.map(buffer50m);
var buffintstreams2 = intstreams2.map(buffer50m);

var buffrails = rails.map(buffer50m);
var buffroadsNE = roadsNE.map(buffer50m);
var buffroadsNW = roadsNW.map(buffer50m);
var buffroadsSE = roadsSE.map(buffer50m);
var buffroadsSW1 = roadsSW1.map(buffer50m);
var buffroadsSW2 = roadsSW2.map(buffer50m);

// Create masks from all the featureCollections
var springsmask = ee.Image.constant(1).clipToCollection(buffsprings).mask().not();
var wellsmask = ee.Image.constant(1).clipToCollection(buffwells).mask().not();
var waterbodiesmask = ee.Image.constant(1).clipToCollection(buffwaterbodies).mask().not();
var perstreamsmask = ee.Image.constant(1).clipToCollection(buffperstreams).mask().not();
var roadsNEmask = ee.Image.constant(1).clipToCollection(buffroadsNE).mask().not();
var roadsNWmask = ee.Image.constant(1).clipToCollection(buffroadsNW).mask().not();
var roadsSEmask = ee.Image.constant(1).clipToCollection(buffroadsSE).mask().not();
var roadsSW1mask = ee.Image.constant(1).clipToCollection(buffroadsSW1).mask().not();
var roadsSW2mask = ee.Image.constant(1).clipToCollection(buffroadsSW2).mask().not();
var railsmask = ee.Image.constant(1).clipToCollection(buffrails).mask().not();
var firesmask = ee.Image.constant(1).clipToCollection(fires).mask().not();
var fires_oldmask = ee.Image.constant(1).clipToCollection(fires_old).mask().not();
var treatmentsmask = ee.Image.constant(1).clipToCollection(treatments).mask().not();
var canalsmask = ee.Image.constant(1).clipToCollection(buffcanals).mask().not();
var intstreams1mask = ee.Image.constant(1).clipToCollection(buffintstreams1).mask().not();
var intstreams2mask = ee.Image.constant(1).clipToCollection(buffintstreams2).mask().not();

//Bring in Landfire EVT data to remove agriculture and human development classes:
//Filter EVT to just our area of interest (limits to the CONUS image from collection)
var evtSage = ee.Image(evt.filterBounds(StudyArea).first());
//Map.addLayer(evtSage, {min:3001, max:4000, palette: 'red,orange,yellow,green,blue,purple'}, "EVT CONUS")

//Remap each veg type:
var remapImage = function(feature) {
 var thisImage = evtSage.remap([feature.get('VALUE')],[feature.get('Reclassify')]).rename('HUMANDEV');
 return thisImage.unmask(2);
};

//Map over each veg type to create ImageCollection of each reclassification:
var evtSage1 = ee.ImageCollection(table.map(remapImage));

//Add all together taking the minimum value at each turn (looking for 1's)
//Note: 2's are only masked areas outside of CONUS
var evtHumanDev = evtSage1.reduce(ee.Reducer.min()).eq(1);
//Convert to a mask
var HumDevMask = evtHumanDev.eq(0);

//Create function to iterate thru masks to create final sampling area
var combineMasks = function(thisMask, firstimg){
  var image = ee.Image(firstimg);
  return image.updateMask(thisMask);
};

//Create list of masks:
var MaskList = ee.ImageCollection(ee.List([springsmask,wellsmask,waterbodiesmask,railsmask,
                                           perstreamsmask,HumDevMask,firesmask,
                                           fires_oldmask,treatmentsmask,canalsmask,
                                           intstreams1mask,intstreams2mask,roadsNEmask,
                                           roadsNWmask,roadsSEmask,roadsSW1mask,roadsSW2mask]));

//Iterate over list to update sageonly and exclude masked areas
var SampleAreas = ee.Image(MaskList.iterate(combineMasks, sageonly));
Map.addLayer(SampleAreas, {min:0, max:1, palette: "#f0f0f0,#000000"}, "Potential Sampling Areas");

//Add climate zones
Map.addLayer(czones, {min: 1, max: 9, palette: '#b35900,#ffff00,#8600b3,#000000,#ff0000,#ff9900,#00b33c,#0099ff,#ff6699'}, "Climate Zones");

//Add ODT
Map.addLayer(odt, {color: 'orange'}, "Oregon Desert Trail");

// Get crs from sagebrush image
var thiscrs = sagebrush.projection().crs();

Export.image.toDrive({
  image:SampleAreas, 
  description: "CandidateSamplingAreas_2km",

  region:exportregion,
  maxPixels:1e13});
